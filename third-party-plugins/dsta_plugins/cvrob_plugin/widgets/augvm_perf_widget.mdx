export const AugmentationFigurePanel = ({ props }) => {
  const resultsData = props.getResults
    ? props.getResults(cid).results
    : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  const AUGMENTATION_LOOKUP = Object.fromEntries(
    ALL_AUGMENTATIONS_LONG.flatMap((aug) =>
      aug.keys.map((key) => [key, aug])
    )
  );

  const panels = resultsData.map((item, index) => {
    const { Augmentation, fig_img, accuracies } = item;

    if (!fig_img) {
      return (
        <div
          key={index}
          style={{
            padding: "1em",
            border: "1px dashed #ccc",
            borderRadius: "8px",
            textAlign: "center",
            color: "#777",
          }}
        >
          <h3>{Augmentation}</h3>
          <div>No image available in sample</div>
        </div>
      );
    }

    let imageUrl = null;

    try {
      imageUrl = props.getArtifactURL(cid, fig_img);
    } catch (error) {
      console.error(`Error loading ${fig_img}`, error);
    }

    const augmentationInfo =
      AUGMENTATION_LOOKUP[Augmentation] ?? {
        longDescription: [
          `No description found for "${Augmentation}".`
        ],
      };
    const acc = accuracies || {};

    const MAX_COLUMNS = 5;

    const severityKeys = Object.keys(acc).sort((a, b) => {
      if (a === "None") return -1;
      if (b === "None") return 1;

      const numA = parseFloat(a.match(/[\d.]+/)?.[0] || 0);
      const numB = parseFloat(b.match(/[\d.]+/)?.[0] || 0);

      return numA - numB;
    });

    const severityChunks = [];
    for (let i = 0; i < severityKeys.length; i += MAX_COLUMNS) {
      severityChunks.push(severityKeys.slice(i, i + MAX_COLUMNS));
    }

    return (
      <div
        key={index}
        style={{
          border: "1px solid #ddd",
          borderRadius: "8px",
          padding: "1.5rem",
          marginBottom: "3rem",
          textAlign: "center",
        }}
      >
        {/* 🔴 Title */}
        <h2
          style={{
            fontWeight: "bold",
            textDecoration: "underline",
            marginBottom: "1rem",
          }}
        >
          Augmentation Name: {Augmentation}
        </h2>

        <h3
          style={{
            fontWeight: "bold",
            textDecoration: "underline",
            marginTop: "1rem",
            marginBottom: "1rem",
          }}
        >
          &gt;&gt; Augmentation Description: &lt;&lt;
        </h3>

        {/* Description */}
        <div style={{ textAlign: "left" }}>
          {augmentationInfo.longDescription.map((p, i) =>
            p.split("\n").map((line, j) =>
              line.trim() === "" ? (
                <br key={`${i}-${j}`} />
              ) : (
                <p key={`${i}-${j}`} style={{ margin: "0.5em 0" }}>
                  {line}
                </p>
              )
            )
          )}
        </div>

        <div
          style={{
            breakInside: "avoid",
            pageBreakInside: "avoid",
            paddingTop: "1rem",
            marginBottom: "1rem",
          }}
        >
          <h3
            style={{
              fontWeight: "bold",
              textDecoration: "underline",
              marginTop: 0,
              marginBottom: "1rem",
            }}
          >
            &gt;&gt; Figure for Augmentation - {Augmentation}: &lt;&lt;
          </h3>

          {/* 📈 Figure */}
          {imageUrl && (
            <img
              src={imageUrl}
              alt={Augmentation}
              style={{
                maxWidth: "700px",
                width: "100%",
                height: "auto",
                borderRadius: "6px",
                marginBottom: "1.5rem",
              }}
            />
          )}

          {/* 📋 Tables */}
          {severityChunks.map((chunk, tableIndex) => (
            <table
              key={tableIndex}
              style={{
                width: "100%",
                borderCollapse: "collapse",
                tableLayout: "fixed",
                marginBottom: "1.5em",
              }}
            >
              <thead>
                <tr>
                  <th style={{ border: "1px solid #ccc", padding: "6px" }}>
                    Severity
                  </th>

                  {chunk.map((sev) => (
                    <th
                      key={sev}
                      style={{
                        border: "1px solid #ccc",
                        padding: "6px",
                        fontSize: "0.85em",
                      }}
                    >
                      {sev === "None" ? "No Aug" : sev}
                    </th>
                  ))}
                </tr>
              </thead>

              <tbody>
                <tr>
                  <td
                    style={{
                      border: "1px solid #ccc",
                      padding: "6px",
                      fontWeight: "bold",
                    }}
                  >
                    Accuracy (%)
                  </td>

                  {chunk.map((sev) => (
                    <td
                      key={sev}
                      style={{
                        border: "1px solid #ccc",
                        padding: "6px",
                        textAlign: "center",
                        fontWeight: "bold",
                      }}
                    >
                      {acc[sev]?.toFixed(1) ?? "-"}
                    </td>
                  ))}
                </tr>
              </tbody>
            </table>
          ))}
          <div style={{ pageBreakAfter: "always" }} />
        </div>
      </div>
    );
  });

  return <div>{panels}</div>;
};

<Header properties={props.properties} />

<AugmentationFigurePanel props={props} />
