export const cid = "augmentation_by_class_algorithm"
//export const ib_cid = "augbyclass_inputblock"


<div style={{ marginBottom: "1rem" }}>
  <div
    style={{
      textAlign: "center",
      fontWeight: "bold",
      fontSize: "2.2rem",
      color: "#b91c1c",
      textDecoration: "underline"
    }}
  >
    {props.properties.title}
  </div>
</div>


import {
  transformData,
  ClassLineChart,
  PopulationChart,
  StackedBarChart,
  ClassMetricsTable
} from './augbyclass-chart.jsx';

import { ALL_AUGMENTATIONS_LONG } from "./augmentationInfo.jsx";

export const AugmentationClassPlots = ({ props }) => {
  const resultsData = props.getResults ? props.getResults(cid).results : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }
  
  const AUGMENTATION_LOOKUP = Object.fromEntries(
    ALL_AUGMENTATIONS_LONG.flatMap((aug) =>
      aug.keys.map((key) => [key, aug])
    )
  );

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: "3rem" }}>
      {resultsData.map((augmentationItem, index) => {
        const { Augmentation, plot_paths } = augmentationItem;
        const augmentationInfo =
          AUGMENTATION_LOOKUP[Augmentation] ?? {
            longDescription: [
              `No description found for "${Augmentation}".`
            ],
          };
        const result = props.getResults(cid);
        const class_names = result.class_names;
        const combined = transformData(
          augmentationItem.classification_report,
          augmentationItem.conf_matrix,
          Object.keys(augmentationItem.display_info),
          class_names
        );
        
        if (!plot_paths || !plot_paths.class_plot) {
          return (
            <div
              key={index}
              style={{
                padding: "1em",
                border: "1px dashed #ccc",
                borderRadius: "8px",
                textAlign: "center",
                color: "#777",
              }}
            >
              <h3>{Augmentation}</h3>
              <div>No image available in sample</div>
            </div>
          );
        }

        const classPlot = plot_paths.class_plot;
        const matplotlibPath = plot_paths.matplotlib_image_path;
        const plotlyPath = plot_paths.plotly_image_path;

        return (
          <div
            key={index}
            style={{
              border: "1px solid #ccc",
              borderRadius: "10px",
              padding: "2rem",
            }}
          >
            {/* 🔴 Augmentation Title */}
            <h2
              style={{
                textAlign: "center",
                fontWeight: "bold",
                textDecoration: "underline",
                color: "red",
                marginBottom: "2rem",
              }}
            >
              Augmentation Name: {Augmentation}
            </h2>

            <h3
              style={{
                textAlign: "center",
                fontWeight: "bold",
                textDecoration: "underline",
                marginTop: "1rem",
                marginBottom: "1rem",
              }}
            >
              &gt;&gt; Augmentation Description: &lt;&lt;
            </h3>

            {/* Description */}
            <div style={{ textAlign: "left" }}>
              {augmentationInfo.longDescription.map((p, i) =>
                p.split("\n").map((line, j) =>
                  line.trim() === "" ? (
                    <br key={`${i}-${j}`} />
                  ) : (
                    <p key={`${i}-${j}`} style={{ margin: "0.5em 0" }}>
                      {line}
                    </p>
                  )
                )
              )}
            </div>

            <div
              style={{
                breakInside: "avoid",
                pageBreakInside: "avoid",
                paddingTop: "1rem",
                marginBottom: "1rem",
              }}
            >

              {/* 🔹 Per Class Section */}
              {Object.entries(classPlot).map(([className, paths], i) => {
                const [metricPath, confmPath, popPath] = paths;

                let metricUrl = null;
                let confmUrl = null;
                let popUrl = null;

                try {
                  metricUrl = props.getArtifactURL(cid, metricPath);
                  confmUrl = props.getArtifactURL(cid, confmPath);
                  popUrl = props.getArtifactURL(cid, popPath);
                } catch (err) {
                  console.error("Artifact URL error:", err);
                }

                return (
                  <div
                    key={i}
                    style={{
                      marginBottom: "3rem",
                      textAlign: "center",
                      breakInside: "avoid",
                      pageBreakInside: "avoid",
                    }}
                  >

                    <h3
                      style={{
                        textAlign: "center",
                        fontWeight: "bold",
                        textDecoration: "underline",
                        marginTop: "1rem",
                        marginBottom: "1rem",
                      }}
                    >
                      &gt;&gt; Figure for Augmentation - {Augmentation} | Class {className} &lt;&lt;
                    </h3>


                    {metricUrl && (
                      <img
                        src={metricUrl}
                        alt={`metric-${className}`}
                        style={{
                          maxWidth: "600px",
                          width: "100%",
                          height: "auto",
                          margin: "0 auto 2rem auto",
                          display: "block",
                        }}
                      />
                    )}

                    <ClassMetricsTable combined={combined} className={className} metrics={["precision", "recall", "f1-score"]} />
                  </div>
                );
              })}
            </div>
          </div>
        );
      })}
    </div>
  );
};

<AugmentationClassPlots props={props} />
