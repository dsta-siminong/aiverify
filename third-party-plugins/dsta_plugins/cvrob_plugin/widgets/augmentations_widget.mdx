export const AugmentationSeverityPanel = ({ props }) => {
  const resultsData = props.getResults ? props.getResults(cid).results : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  const augmentationPanels = resultsData.map((augmentationItem, index) => {
    const { Augmentation, display_info } = augmentationItem;

    if (!display_info) return null;

    // Sort severity keys properly
    const severityKeys = Object.keys(display_info).sort((a, b) => {
      if (a === "severityNone") return -1;
      if (b === "severityNone") return 1;

      const numA = parseFloat(a.match(/[\d.]+/)?.[0] || 0);
      const numB = parseFloat(b.match(/[\d.]+/)?.[0] || 0);

      return numA - numB;
    });

    const severityEntries = severityKeys.map((severityKey) => {
      const [imagePath, groundTruth, prediction] =
        display_info[severityKey] || [];

      if (!imagePath) return null;

      let imageUrl = null;

      try {
        imageUrl = props.getArtifactURL(cid, imagePath);
      } catch (error) {
        console.error(`Error getting URL for ${imagePath}`, error);
      }

      // Extract filename from path
      const fileName = imagePath.split("/").pop();

      return {
        severityKey,
        imageUrl,
        groundTruth,
        prediction,
        fileName,
      };
    });

    return (
      <div
        key={index}
        style={{
          border: "1px solid #ddd",
          borderRadius: "8px",
          padding: "1.5rem",
          marginBottom: "3rem",
        }}
      >
        {/* 🔴 Centered Bold Underlined Red Title */}
        <h2
          style={{
            textAlign: "center",
            fontWeight: "bold",
            textDecoration: "underline",
            color: "red",
            marginBottom: "0.5rem",
          }}
        >
          {Augmentation}
        </h2>

        {/* Subtitle: filename */}
        {severityEntries[0] && (
          <div
            style={{
              textAlign: "center",
              marginBottom: "1.5rem",
              fontStyle: "italic",
              color: "#666",
            }}
          >
            {severityEntries[0].fileName}
          </div>
        )}

        <div
          style={{
            display: "flex",
            flexWrap: "wrap",
            gap: "2rem",
            justifyContent: "center",
          }}
        >
          {severityEntries.map((entry, i) => {
            if (!entry) return null;

            const isCorrect = entry.groundTruth === entry.prediction;

            return (
              <div key={i} style={{ textAlign: "center" }}>
                <h4>{entry.severityKey}</h4>

                <img
                  src={entry.imageUrl}
                  alt={entry.severityKey}
                  style={{
                    maxWidth: "300px",   // 👈 no fixed width/height
                    height: "auto",      // 👈 keeps original aspect ratio
                    borderRadius: "6px",
                    border: isCorrect
                      ? "3px solid #4CAF50"
                      : "3px solid #F44336",
                  }}
                />

                <div style={{ marginTop: "0.5rem" }}>
                  <div>
                    <strong>GT:</strong> {entry.groundTruth}
                  </div>
                  <div>
                    <strong>Pred:</strong>{" "}
                    <span
                      style={{
                        color: isCorrect ? "#4CAF50" : "#F44336",
                        fontWeight: "bold",
                      }}
                    >
                      {entry.prediction}
                    </span>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      </div>
    );
  });

  return <div>{augmentationPanels}</div>;
};

<AugmentationSeverityPanel props={props} />


export const AugmentationFigurePanel = ({ props }) => {
  const resultsData = props.getResults
    ? props.getResults(cid).results
    : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  const panels = resultsData.map((item, index) => {
    const { Augmentation, fig_img, accuracies } = item;

    if (!fig_img) return <div key={index}>No FIGIMG</div>;

    let imageUrl = null;

    try {
      imageUrl = props.getArtifactURL(algo_cid, fig_img);
    } catch (error) {
      console.error(`Error loading ${fig_img}`, error);
    }

    const acc = accuracies || {};

    // ✅ Sort severities properly
    const severityKeys = Object.keys(acc).sort((a, b) => {
      if (a === "None") return -1;
      if (b === "None") return 1;

      const numA = parseFloat(a.match(/[\d.]+/)?.[0] || 0);
      const numB = parseFloat(b.match(/[\d.]+/)?.[0] || 0);

      return numA - numB;
    });

    return (
      <div
        key={index}
        style={{
          border: "1px solid #ddd",
          borderRadius: "8px",
          padding: "1.5rem",
          marginBottom: "3rem",
          textAlign: "center",
        }}
      >
        {/* 🔴 Title */}
        <h2
          style={{
            fontWeight: "bold",
            textDecoration: "underline",
            color: "red",
            marginBottom: "1rem",
          }}
        >
          {Augmentation}
        </h2>

        {/* 📈 Figure */}
        {imageUrl && (
          <img
            src={imageUrl}
            alt={Augmentation}
            style={{
              maxWidth: "700px",
              width: "100%",
              height: "auto",
              borderRadius: "6px",
              marginBottom: "1.5rem",
            }}
          />
        )}

        {/* 📋 Table */}
        <table
          style={{
            width: "100%",
            borderCollapse: "collapse",
            tableLayout: "fixed",
          }}
        >
          <thead>
            <tr>
              {severityKeys.map((sev) => (
                <th
                  key={sev}
                  style={{
                    border: "1px solid #ccc",
                    padding: "6px",
                    fontSize: "0.85em",
                  }}
                >
                  {sev === "None" ? "No Aug" : sev}
                </th>
              ))}
            </tr>
          </thead>

          <tbody>
            <tr>
              {severityKeys.map((sev) => (
                <td
                  key={sev}
                  style={{
                    border: "1px solid #ccc",
                    padding: "6px",
                    textAlign: "center",
                    fontWeight: "bold",
                  }}
                >
                  {acc[sev]?.toFixed(1) ?? "-"}
                </td>
              ))}
            </tr>
          </tbody>
        </table>
      </div>
    );
  });

  return <div>{panels}</div>;
};
