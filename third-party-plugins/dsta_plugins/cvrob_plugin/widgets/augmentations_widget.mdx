export const AugmentationFigurePanel = ({ props }) => {
  const resultsData = props.getResults
    ? props.getResults(cid).results
    : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  const panels = resultsData.map((item, index) => {
    const { Augmentation, fig_img, accuracies } = item;

    if (!fig_img) return <div key={index}>No FIGIMG</div>;

    let imageUrl = null;

    try {
      imageUrl = props.getArtifactURL(cid, fig_img);
    } catch (error) {
      console.error(`Error loading ${fig_img}`, error);
    }

    const acc = accuracies || {};

    // ✅ Sort severities properly
    const severityKeys = Object.keys(acc).sort((a, b) => {
      if (a === "None") return -1;
      if (b === "None") return 1;

      const numA = parseFloat(a.match(/[\d.]+/)?.[0] || 0);
      const numB = parseFloat(b.match(/[\d.]+/)?.[0] || 0);

      return numA - numB;
    });

    return (
      <div
        key={index}
        style={{
          border: "1px solid #ddd",
          borderRadius: "8px",
          padding: "1.5rem",
          marginBottom: "3rem",
          textAlign: "center",
        }}
      >
        {/* 🔴 Title */}
        <h2
          style={{
            fontWeight: "bold",
            textDecoration: "underline",
            color: "red",
            marginBottom: "1rem",
          }}
        >
          {Augmentation}
        </h2>

        {/* 📈 Figure */}
        {imageUrl && (
          <img
            src={imageUrl}
            alt={Augmentation}
            style={{
              maxWidth: "700px",
              width: "100%",
              height: "auto",
              borderRadius: "6px",
              marginBottom: "1.5rem",
            }}
          />
        )}

        {/* 📋 Table */}
        <table
          style={{
            width: "100%",
            borderCollapse: "collapse",
            tableLayout: "fixed",
          }}
        >
          <thead>
            <tr>
              {severityKeys.map((sev) => (
                <th
                  key={sev}
                  style={{
                    border: "1px solid #ccc",
                    padding: "6px",
                    fontSize: "0.85em",
                  }}
                >
                  {sev === "None" ? "No Aug" : sev}
                </th>
              ))}
            </tr>
          </thead>

          <tbody>
            <tr>
              {severityKeys.map((sev) => (
                <td
                  key={sev}
                  style={{
                    border: "1px solid #ccc",
                    padding: "6px",
                    textAlign: "center",
                    fontWeight: "bold",
                  }}
                >
                  {acc[sev]?.toFixed(1) ?? "-"}
                </td>
              ))}
            </tr>
          </tbody>
        </table>
      </div>
    );
  });

  return <div>{panels}</div>;
};
