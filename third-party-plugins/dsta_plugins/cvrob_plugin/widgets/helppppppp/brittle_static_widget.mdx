export const BrittleDisplay = ({ props }) => {
  const key = Object.keys(props.artifacts || {})[0];
  const artifactList = key ? props.artifacts[key] : [];

  // 🔥 Keep only main plot artifacts
  const filteredArtifacts = artifactList.filter((url) =>
    url.includes("/matplotlib/") || url.includes("/plotly/")
  );

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: "4rem" }}>
      {filteredArtifacts.map((url, i) => {
        const filename = url.split("/").pop();

        const title = filename
          ?.replace(".png", "")
          ?.replace(".html", "")
          ?.replace(/_/g, " ");

        return (
          <div
            key={i}
            style={{
              padding: "1.5rem",
              background: "#fff",
              borderRadius: "12px",
              boxShadow: "0 2px 8px rgba(0,0,0,0.06)"
            }}
          >
            <h3 style={{ marginBottom: "1rem" }}>{title}</h3>

            {url.endsWith(".png") && (
              <img
                src={url}
                style={{
                  width: "100%",
                  borderRadius: "8px",
                }}
              />
            )}

            {url.endsWith(".html") && url.indexOf("top_k" > -1) && (
              <iframe
                src={url}
                style={{
                  width: "100%",
                  height: "800px",
                  border: "none",
                }}
              />
            )}
          </div>
        );
      })}
    </div>
  );
};

<BrittleDisplay props={props} />
