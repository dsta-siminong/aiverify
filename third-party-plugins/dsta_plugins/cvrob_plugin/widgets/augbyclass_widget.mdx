export const AugmentationSeverityPanel = ({ props }) => {
  const resultsData = props.getResults ? props.getResults(cid).results : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  const augmentationPanels = resultsData.map((augmentationItem, index) => {
    const { Augmentation, display_info } = augmentationItem;

    if (!display_info) return null;

    // Sort severity keys properly
    const severityKeys = Object.keys(display_info).sort((a, b) => {
      if (a === "severityNone") return -1;
      if (b === "severityNone") return 1;

      const numA = parseFloat(a.match(/[\d.]+/)?.[0] || 0);
      const numB = parseFloat(b.match(/[\d.]+/)?.[0] || 0);

      return numA - numB;
    });

    const severityEntries = severityKeys.map((severityKey) => {
      const [imagePath, groundTruth, prediction] =
        display_info[severityKey] || [];

      if (!imagePath) return null;

      let imageUrl = null;

      try {
        imageUrl = props.getArtifactURL(cid, imagePath);
      } catch (error) {
        console.error(`Error getting URL for ${imagePath}`, error);
      }

      // Extract filename from path
      const fileName = imagePath.split("/").pop();

      return {
        severityKey,
        imageUrl,
        groundTruth,
        prediction,
        fileName,
      };
    });

    return (
      <div
        key={index}
        style={{
          border: "1px solid #ddd",
          borderRadius: "8px",
          padding: "1.5rem",
          marginBottom: "3rem",
        }}
      >
        {/* 🔴 Centered Bold Underlined Red Title */}
        <h2
          style={{
            textAlign: "center",
            fontWeight: "bold",
            textDecoration: "underline",
            color: "red",
            marginBottom: "0.5rem",
          }}
        >
          {Augmentation}
        </h2>

        {/* Subtitle: filename */}
        {severityEntries[0] && (
          <div
            style={{
              textAlign: "center",
              marginBottom: "1.5rem",
              fontStyle: "italic",
              color: "#666",
            }}
          >
            {severityEntries[0].fileName}
          </div>
        )}

        <div
          style={{
            display: "flex",
            flexWrap: "wrap",
            gap: "2rem",
            justifyContent: "center",
          }}
        >
          {severityEntries.map((entry, i) => {
            if (!entry) return null;

            const isCorrect = entry.groundTruth === entry.prediction;

            return (
              <div key={i} style={{ textAlign: "center" }}>
                <h4>{entry.severityKey}</h4>

                <img
                  src={entry.imageUrl}
                  alt={entry.severityKey}
                  style={{
                    maxWidth: "300px",   // 👈 no fixed width/height
                    height: "auto",      // 👈 keeps original aspect ratio
                    borderRadius: "6px",
                    border: isCorrect
                      ? "3px solid #4CAF50"
                      : "3px solid #F44336",
                  }}
                />

                <div style={{ marginTop: "0.5rem" }}>
                  <div>
                    <strong>GT:</strong> {entry.groundTruth}
                  </div>
                  <div>
                    <strong>Pred:</strong>{" "}
                    <span
                      style={{
                        color: isCorrect ? "#4CAF50" : "#F44336",
                        fontWeight: "bold",
                      }}
                    >
                      {entry.prediction}
                    </span>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      </div>
    );
  });

  return <div>{augmentationPanels}</div>;
};

<AugmentationSeverityPanel props={props} />

export const AugmentationClassPlots = ({ props }) => {
  const resultsData = props.getResults ? props.getResults(cid).results : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: "3rem" }}>
      {resultsData.map((augmentationItem, index) => {
        const { Augmentation, plot_paths } = augmentationItem;
        const result = props.getResults(cid);
        const class_names = result.class_names;
        const combined = transformData(
          augmentationItem.classification_report,
          augmentationItem.conf_matrix,
          Object.keys(augmentationItem.display_info),
          class_names
        );
        
        if (!plot_paths || !plot_paths.class_plot) return null;

        const classPlot = plot_paths.class_plot;
        const matplotlibPath = plot_paths.matplotlib_image_path;
        const plotlyPath = plot_paths.plotly_image_path;

        return (
          <div
            key={index}
            style={{
              border: "1px solid #ccc",
              borderRadius: "10px",
              padding: "2rem",
            }}
          >
            {/* 🔴 Augmentation Title */}
            <h2
              style={{
                textAlign: "center",
                fontWeight: "bold",
                textDecoration: "underline",
                color: "red",
                marginBottom: "2rem",
              }}
            >
              {Augmentation}
            </h2>

            {/* 🔹 Per Class Section */}
            {Object.entries(classPlot).map(([className, paths], i) => {
              const [metricPath, confmPath, popPath] = paths;

              let metricUrl = null;
              let confmUrl = null;
              let popUrl = null;

              try {
                metricUrl = props.getArtifactURL(cid, metricPath);
                confmUrl = props.getArtifactURL(cid, confmPath);
                popUrl = props.getArtifactURL(cid, popPath);
              } catch (err) {
                console.error("Artifact URL error:", err);
              }

              return (
                <div
                  key={i}
                  style={{
                    marginBottom: "3rem",
                    textAlign: "center",
                  }}
                >
                  <h3 style={{ marginBottom: "1rem" }}>
                    Class {className}
                  </h3>

                  {/* Metric Section */}
                  <div style={{ marginBottom: "1rem", fontStyle: "italic" }}>
                    Metric for Class {className}
                  </div>

                  {metricUrl && (
                    <img
                      src={metricUrl}
                      alt={`metric-${className}`}
                      style={{
                        maxWidth: "600px",
                        width: "100%",
                        height: "auto",
                        margin: "0 auto 2rem auto",
                        display: "block",
                      }}
                    />
                  )}

                  <ClassMetricsTable combined={combined} className={className} metrics={["precision", "recall", "f1-score"]} />

                  {/* CM Section */}
                  <div style={{ marginBottom: "1rem", fontStyle: "italic" }}>
                    Confusion Matrix values for Class {className}
                  </div>

                  {confmUrl && (
                    <img
                      src={confmUrl}
                      alt={`confm-${className}`}
                      style={{
                        maxWidth: "600px",
                        width: "100%",
                        height: "auto",
                        margin: "0 auto 2rem auto",
                        display: "block",
                      }}
                    />
                  )}

                  <ClassMetricsTable combined={combined} className={className} metrics={["TP", "FP", "FN", "TN"]} />

                  {/* Population Section */}
                  <div style={{ marginBottom: "1rem", fontStyle: "italic" }}>
                    Populations for Class {className}
                  </div>

                  {popUrl && (
                    <img
                      src={popUrl}
                      alt={`pop-${className}`}
                      style={{
                        maxWidth: "600px",
                        width: "100%",
                        height: "auto",
                        margin: "0 auto 2rem auto",
                        display: "block",
                      }}
                    />
                  )}

                  <ClassMetricsTable combined={combined} className={className} metrics={["preds_population", "actual_population"]} />
                </div>
              );
            })}

            {/* 🔹 Final Matplotlib Summary Image */}
            {matplotlibPath && (
              <div style={{ textAlign: "center", marginTop: "2rem" }}>
                <h3 style={{ marginBottom: "1rem" }}>
                  Overall Matplotlib Summary for {Augmentation}
                </h3>

                <img
                  src={props.getArtifactURL(cid, matplotlibPath)}
                  alt="matplotlib-summary"
                  style={{
                    maxWidth: "800px",
                    width: "100%",
                    height: "auto",
                    margin: "0 auto",
                    display: "block",
                  }}
                />
              </div>
            )}

            {/* 🔹 Final Plotly Summary (HTML) */}
            {plotlyPath && (
              <div style={{ textAlign: "center", marginTop: "2rem" , overflowX: "auto"}}>
                <h3 style={{ marginBottom: "1rem" }}>
                  Overall Plotly Summary for {Augmentation}
                </h3>

                <iframe
                  src={props.getArtifactURL(cid, plotlyPath)}
                  title={`plotly-summary-${Augmentation}`}
                  style={{
                    width: "100%",
                    maxWidth: "1000px",
                    height: "700px",
                    border: "none",
                    display: "block",
                    margin: "0 auto",
                  }}
                />
              </div>
            )}
          </div>
        );
      })}
    </div>
  );
};

<AugmentationClassPlots props={props} />

export const ConfusionMatricesSection = ({ props }) => {
  const resultsData = props.getResults ? props.getResults(cid).results : null;

  if (!resultsData || resultsData.length === 0) {
    return <div>No results data available</div>;
  }

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: "3rem" }}>
      <h2
        style={{
          textAlign: "center",
          fontWeight: "bold",
          textDecoration: "underline",
          color: "red",
          marginBottom: "2rem",
        }}
      >
        Confusion Matrices
      </h2>

      {resultsData.map((augmentationItem, index) => {
        const { Augmentation, confusion_matrix } = augmentationItem;

        if (!confusion_matrix) return null;

        return (
          <div
            key={index}
            style={{
              border: "1px solid #ccc",
              borderRadius: "10px",
              padding: "2rem",
            }}
          >
            <h3
              style={{
                textAlign: "center",
                fontWeight: "bold",
                marginBottom: "2rem",
              }}
            >
              {Augmentation}
            </h3>

            {Object.entries(confusion_matrix).map(([severity, paths], i) => {
              const [pngPath, htmlPath] = paths;

              let pngUrl = null;
              let htmlUrl = null;

              try {
                pngUrl = props.getArtifactURL(cid, pngPath);
                htmlUrl = props.getArtifactURL(cid, htmlPath);
              } catch (err) {
                console.error("Artifact URL error:", err);
              }

              return (
                <div key={i} style={{ marginBottom: "3rem" }}>
                  <h4 style={{ marginBottom: "1rem" }}>{severity}</h4>

                  {pngUrl && (
                    <img
                      src={pngUrl}
                      alt={`Confusion Matrix PNG - ${severity}`}
                      style={{
                        maxWidth: "600px",
                        width: "100%",
                        display: "block",
                        marginBottom: "1rem",
                      }}
                    />
                  )}

                  {htmlUrl && (
                    <iframe
                      src={htmlUrl}
                      title={`Confusion Matrix HTML - ${severity}`}
                      style={{
                        width: "100%",
                        height: "500px",
                        border: "1px solid #ccc",
                        borderRadius: "6px",
                      }}
                    />
                  )}
                </div>
              );
            })}
          </div>
        );
      })}
    </div>
  );
};
